    10	import { platform } from "node:os";

    12	const WANT = "3.13";

    14	/** "3.13" for an interpreter command, or null when it cannot be run. */
    15	function versionOf(cmd) {
    16	  try {
    17	    return execSync(`${cmd} -c "import sys; print('%d.%d' % sys.version_info[:2])"`, {
    18	      encoding: "utf8",
    19	      stdio: ["ignore", "pipe", "ignore"],
    20	    }).trim();
    21	  } catch {
    22	    return null;
    23	  }
    24	}

    26	/**
    27	 * The command for a Python 3.13 interpreter: `VYSTED_PYTHON` when set, else the
    28	 * first of `python3.13`, `py -3.13` (Windows) or a plain `python3`/`python`
    29	 * that reports 3.13. Throws a message naming the fix when none does.
    30	 */
    31	export function resolveBuildPython() {
    32	  const override = process.env.VYSTED_PYTHON;
    33	  if (override) {
    34	    const cmd = `"${override}"`;
    35	    const got = versionOf(cmd);
    36	    if (got !== WANT) {
    37	      throw new Error(
    38	        `VYSTED_PYTHON=${override} is Python ${got ?? "(not runnable)"}, need ${WANT}.`,
    39	      );
    40	    }
    41	    return cmd;
    42	  }
    43	  const candidates =
    44	    platform() === "win32" ? ["py -3.13", "python3.13", "python"] : ["python3.13", "python3"];
    45	  for (const cmd of candidates) {
    46	    if (versionOf(cmd) === WANT) return cmd;
    47	  }
    48	  throw new Error(
    49	    `No Python ${WANT} found (tried ${candidates.join(", ")}). Install Python ${WANT} ` +
    50	      `(e.g. \`brew install python@3.13\`) or set VYSTED_PYTHON to its path.`,
    51	  );
    52	}

    54	/**
    55	 * Make `venvDir` a Python 3.13 venv: create it when missing and recreate it
    56	 * when its interpreter is any other version.
    57	 */
    58	export function ensureBuildVenv(venvDir, venvPython, run) {
    59	  if (existsSync(venvPython)) {
    60	    const got = versionOf(`"${venvPython}"`);
    61	    if (got === WANT) return;

    63	    rmSync(venvDir, { recursive: true, force: true });
    64	  }
    65	  run(`${resolveBuildPython()} -m venv "${venvDir}"`);
