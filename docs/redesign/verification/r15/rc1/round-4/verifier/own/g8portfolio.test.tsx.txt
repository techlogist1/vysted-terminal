import { describe, expect, it, vi } from "vitest";
import { act, fireEvent, render, screen, waitFor } from "@testing-library/react";
import * as fs from "node:fs";
import * as path from "node:path";

const OUT = process.env.G8_OUT as string;
const PORT = process.env.G8_PORT as string;
const LOG = (...a: unknown[]) => fs.appendFileSync(process.env.G8_LOG as string, a.map(String).join(" ") + "\n");
const calls: Array<{ cmd: string; args: unknown }> = [];
vi.mock("@tauri-apps/api/core", () => ({
  invoke: vi.fn(async (cmd: string, args: any) => {
    calls.push({ cmd, args });
    if (cmd === "get_sidecar_port") return { port: Number(PORT), state: "ready", reason: null };
    if (cmd === "get_app_data_dir") return OUT;
    if (cmd === "write_text_atomic") {
      fs.mkdirSync(path.dirname(args.path), { recursive: true });
      fs.writeFileSync(args.path, args.contents);
      return null;
    }
    return null;
  }),
}));

import { usePortfoliosStore } from "@/store/portfolios";
import { useSettingsStore } from "@/store/settings";
import { PortfolioPanel } from "@/modules/portfolio/PortfolioPanel";

describe("gate8 tracked portfolio round trip (verifier)", () => {
  it("add with cost basis -> live P&L -> CSV export -> delete", async () => {
    (window as any).__TAURI_INTERNALS__ = {};
    usePortfoliosStore.setState({ portfolios: [{ id: "default", name: "Portfolio", holdings: [] }], activeId: "default" });
    useSettingsStore.setState({ region: "US" });
    const live = await (await fetch(`http://127.0.0.1:${PORT}/quotes/MSFT?asset_class=equity`)).json();
    LOG("LIVE_QUOTE", JSON.stringify({ symbol: live.symbol, price: live.price, currency: live.currency, provider: live.provider, timestamp: live.timestamp }));
    render(<PortfolioPanel />);
    fireEvent.change(screen.getByLabelText("Symbol"), { target: { value: "msft" } });
    fireEvent.change(screen.getByLabelText("Quantity"), { target: { value: "10" } });
    fireEvent.change(screen.getByLabelText("Avg cost / share"), { target: { value: "300" } });
    await act(async () => { fireEvent.submit(screen.getByLabelText("Symbol").closest("form")!); });
    const h = usePortfoliosStore.getState().portfolios[0].holdings;
    LOG("STORE_AFTER_ADD", JSON.stringify(h));
    expect(h).toHaveLength(1);
    expect(h[0]).toMatchObject({ symbol: "MSFT", quantity: 10, costBasis: 300 });
    const expPnl = (live.price * 10 - 3000);
    await waitFor(() => expect(document.body.textContent).toMatch(/\+\$[\d,]+\.\d\d \(\+[\d.]+%\)|-\$[\d,]+\.\d\d/), { timeout: 45000 });
    const m = (document.body.textContent || "").match(/[+-]\$[\d,]+\.\d\d \([+-][\d.]+%\)/);
    LOG("PNL_RENDERED", m && m[0], "EXPECTED_PNL", expPnl.toFixed(2));
    await act(async () => { fireEvent.click(screen.getByLabelText("Export portfolio to CSV")); });
    await waitFor(() => expect(calls.some((c) => c.cmd === "write_text_atomic")).toBe(true), { timeout: 15000 });
    const w = calls.find((c) => c.cmd === "write_text_atomic")!.args as any;
    LOG("CSV_PATH", w.path);
    LOG("CSV_CONTENT\n" + w.contents);
    const row = w.contents.split("\n")[1].split(",");
    expect(row[0]).toBe("MSFT");
    expect(Number(row[7])).toBeCloseTo(Number(row[6]) - 3000, 6);
    await waitFor(() => expect(document.body.textContent).toMatch(/Saved .*vysted-portfolio-portfolio\.csv/));
    const del = screen.getByLabelText("Delete MSFT");
    await act(async () => { fireEvent.click(del); });
    LOG("STORE_AFTER_FIRST_CLICK(armed)", usePortfoliosStore.getState().portfolios[0].holdings.length);
    await act(async () => { fireEvent.click(del); });
    const after = usePortfoliosStore.getState().portfolios[0].holdings;
    LOG("STORE_AFTER_DELETE", JSON.stringify(after));
    expect(after).toHaveLength(0);
    LOG("INVOKE_CMDS", JSON.stringify(calls.map((c) => c.cmd)));
  });
});
