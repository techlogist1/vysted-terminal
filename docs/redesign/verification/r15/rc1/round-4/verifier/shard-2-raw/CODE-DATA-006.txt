describe("vshard-2 fresh cases (R15-CODE-DATA-006)", () => {
  it("fresh: run A's STREAM result frame lands after run B finished -> B's result/status untouched", async () => {
    const RESULT_A: ScreenerResult = { ...RESULT_SAMPLE, universe: "nifty50", rows: [] };
    const encoder = new TextEncoder();
    let pushA!: () => void;
    const streamA = new ReadableStream({
      start(c) {
        c.enqueue(encoder.encode(`data: ${JSON.stringify({ event: "progress", phase: "sweep", done: 1, total: 9, detail: "a" })}\n\n`));
        pushA = () => {
          c.enqueue(encoder.encode(`data: ${JSON.stringify({ event: "result", ...RESULT_A })}\n\n`));
          c.close();
        };
      },
    });
    let calls = 0;
    vi.spyOn(globalThis, "fetch").mockImplementation(async () => {
      calls++;
      if (calls === 1) return new Response(streamA, { status: 200, headers: { "Content-Type": "text/event-stream" } });
      return makeStreamResponse(RESULT_SAMPLE);
    });
    const runA = useScreenerStore.getState().runScreener();
    await vi.waitFor(() => expect(calls).toBe(1));
    await useScreenerStore.getState().runScreener();
    pushA();
    expect(await runA).toBeNull();
    expect(useScreenerStore.getState().lastResult).toEqual(RESULT_SAMPLE);
    expect(useScreenerStore.getState().status).toBe("ready");
    expect(useScreenerStore.getState().progress).toBeNull();
  });

  it("adjacent probe: a run rejected by the formula pre-flight while run A is in flight — does A's late result overwrite the formula error?", async () => {
    let resolveA!: (r: Response) => void;
    const pending = new Promise<Response>((r) => (resolveA = r));
    vi.spyOn(globalThis, "fetch").mockImplementation(async () => pending);
    const runA = useScreenerStore.getState().runScreener();
    await vi.waitFor(() => expect(useScreenerStore.getState().status).toBe("loading"));
    useScreenerStore.getState().setFormula("pe < (");
    await useScreenerStore.getState().runScreener();
    const afterPreflight = { status: useScreenerStore.getState().status, error: useScreenerStore.getState().error };
    resolveA(makeStreamResponse(RESULT_SAMPLE));
    await runA;
    const final = { status: useScreenerStore.getState().status, error: useScreenerStore.getState().error, hasResult: useScreenerStore.getState().lastResult !== null };
    expect({ afterPreflight, final }).toEqual({ afterPreflight, final: { ...afterPreflight, hasResult: false } });
  });
});
