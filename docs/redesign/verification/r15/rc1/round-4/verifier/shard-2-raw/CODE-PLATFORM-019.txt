from services import workflow_engine as we
from models.workflow import WorkflowSpec
T0 = time.monotonic(); log = []
def mk(name, delay, out=1):

        await asyncio.sleep(delay)
        log.append((name, "end", round(time.monotonic() - T0, 2)))
