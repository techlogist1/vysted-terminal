=== openbb_mcp_subprocess/requirements.txt ===
# openbb-mcp subprocess — its own venv with OpenBB's strict pins.
#
# openbb-mcp-server 1.4.0 pulls in openbb-core 1.6.x, which strictly pins
# fastapi (<0.129) and uvicorn (<0.41). Those pins are incompatible with
# the main Vysted sidecar's pins (fastapi 0.136, uvicorn 0.46), so this
# subprocess gets its own venv and its own PyInstaller --onefile binary
# built by ``scripts/ensure-openbb-mcp-sidecar.mjs``.
#
# This is the architectural fix for the v0.3.0 Phase-2 OpenBB deadlock:
# the bespoke FastAPI subprocess is replaced by the stock openbb-mcp-server
# package, spawned via Tauri Rust Command::new (different Windows handle
# semantics than Python subprocess.Popen — see CLAUDE.md Gotcha). The main
# Vysted sidecar talks to this child as an MCP client (Streamable-HTTP).
openbb-mcp-server==1.4.0
# OpenBB Platform extensions register REST routes via importlib entry-points
# at runtime; openbb-mcp-server reads those routes and exposes them as MCP
# tools. Without the extensions, the server only exposes its meta-tools
# (skill management, prompt discovery). Each extension pulled here adds a
# category of OpenBB data tools (equity_price_quote, equity_fundamental_*,
# economy_fred_series, etc.).
openbb-equity==1.6.1
openbb-economy==1.6.1
openbb-yfinance==1.6.2
openbb-fred==1.6.0
openbb-fmp==1.6.0
openbb-news==1.6.0


# The MCP stack, pinned to the known-good freeze (R15-LEAD-001): unpinned, a
# fresh venv resolved fastmcp 4.x / an mcp built on httpx2, and the build's
# --copy-metadata=httpx then failed. Bump these together, deliberately.
fastmcp==3.3.1
fastmcp-slim==3.3.1
mcp==1.27.1
httpx==0.28.1

=== sec_edgar_mcp_subprocess/requirements.txt ===
# sec-edgar-mcp subprocess — its own venv so we can pin the upstream
# SEC EDGAR MCP server independently of the main Vysted sidecar.
#
# The package is pure-Python (httpx + a thin XBRL parser); it has no
# transitive pin conflicts with the main sidecar's fastapi / uvicorn /
# pydantic stack today, but isolating it preserves the v0.4.0 pattern
# (precedent: openbb-mcp-server). When sec-edgar-mcp 1.x bumps a shared
# dependency we want it confined to this venv, not the main sidecar.
#
# This subprocess is packaged via ``scripts/ensure-sec-edgar-mcp-sidecar.mjs``
# into a PyInstaller --onefile binary and spawned by Tauri Rust
# ``Command::new`` from ``src-tauri/src/sec_edgar_mcp.rs`` — the v0.4.0
# Phase-2 deadlock fix pattern.
sec-edgar-mcp==1.0.8

# The MCP stack, pinned to the known-good freeze (R15-LEAD-001): unpinned, a
# fresh venv resolved an mcp built on httpx2, and the build's
# --copy-metadata=httpx then failed. Bump these together, deliberately.
mcp==1.27.1
httpx==0.28.1
