import { beforeEach, describe, expect, it, vi } from "vitest";
vi.mock("@/lib/sidecar-client", async (importOriginal) => ({
  ...(await importOriginal<typeof import("@/lib/sidecar-client")>()),
  getSidecarBaseUrl: vi.fn(async () => "http://127.0.0.1:8787"),
}));
vi.mock("@/lib/search-headers", () => ({ buildSearchHeaders: vi.fn(async () => ({})) }));
import { streamAgentInvocation, streamChat } from "./streaming";

const live422 = '{"detail":[{"type":"missing","loc":["body","provider"],"msg":"Field required","input":{}},{"type":"missing","loc":["body","model"],"msg":"Field required","input":{}},{"type":"missing","loc":["body","messages"],"msg":"Field required","input":{}}]}';
async function errOf(fn: (h: { onEvent: () => void; onError: (e: Error) => void }) => Promise<void>) {
  let got: Error | undefined;
  await fn({ onEvent: () => undefined, onError: (e) => { got = e; } });
  return got;
}
describe("vs3 fresh UI-012", () => {
  beforeEach(() => vi.unstubAllGlobals());
  it("raw-chat 422 array -> field: msg sentences, no JSON", async () => {
    vi.stubGlobal("fetch", vi.fn(async () => new Response(live422, { status: 422 })));
    const e = await errOf((h) => streamChat({ provider: "x", model: "y", messages: [] } as never, h));
    console.log("422 ->", e?.message);
    expect(e?.message).toBe("provider: Field required; model: Field required; messages: Field required");
    expect(e?.message).not.toContain("{");
  });
  it("agent 404 string detail", async () => {
    vi.stubGlobal("fetch", vi.fn(async () => new Response('{"detail":"unknown agent: \'nope\'"}', { status: 404 })));
    const e = await errOf((h) => streamAgentInvocation("nope", { prompt: "hi" }, h));
    console.log("404 ->", e?.message);
    expect(e?.message).toBe("unknown agent: 'nope'");
  });
  it("non-JSON 500 (proxy HTML) -> fallback sentence, no HTML", async () => {
    vi.stubGlobal("fetch", vi.fn(async () => new Response("<html><body>Internal Server Error</body></html>", { status: 500 })));
    const e = await errOf((h) => streamAgentInvocation("copilot", { prompt: "hi" }, h));

    expect(e?.message).not.toContain("<html>");
  });
  it("engine dies mid-stream -> unreachable sentence", async () => {
    const body = new ReadableStream<Uint8Array>({
      start(c) { c.enqueue(new TextEncoder().encode('data: {"kind":"token","text":"hel"}\n\n')); },

    });

    console.log("midstream ->", e?.message);
    expect(e?.message).toContain("not responding");
  });
});
