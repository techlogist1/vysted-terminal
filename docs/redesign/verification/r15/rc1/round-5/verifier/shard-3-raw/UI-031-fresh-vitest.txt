describe("vs3 fresh UI-031", () => {
  it("A: three commands, loads settle in reverse; a superseded REJECTION never shows and the agent context names only the newest", async () => {
    const pending: Record<string, { res: (v: EquityOverview) => void; rej: (e: unknown) => void }> = {};
    mockLoad.mockImplementation(
      (symbol: string) => new Promise((res, rej) => (pending[symbol] = { res, rej })),
    );
    render(<EquityOverviewPanel />);
    for (const symbol of ["MSFT", "INFY.NS", "NVDA"]) {
      await act(async () => {
        useEquityCommandStore.getState().loadSymbol(symbol);
      });
      await flushCommandTick();
    }
    await act(async () => {
      pending["NVDA"]!.res(overview({ symbol: "NVDA", fundamentals: fundamentals({ name: "NVIDIA Corp" }) }));
    });
    expect(screen.getByText("NVIDIA Corp")).toBeInTheDocument();
    await act(async () => {
      pending["INFY.NS"]!.rej(new SidecarError(502, "infy upstream down"));
    });
    await act(async () => {
      pending["MSFT"]!.res(overview({ symbol: "MSFT", fundamentals: fundamentals({ name: "Microsoft" }) }));
    });
    expect(screen.getByText("NVIDIA Corp")).toBeInTheDocument();
    expect(screen.queryByText("Microsoft")).toBeNull();
    expect(screen.queryByText(/infy upstream down/)).toBeNull();
    const events = Object.values(usePanelContextBus.getState().lastEventBySource);
    const tickers = events.map((e) => (e.payload as { ticker?: string }).ticker).filter(Boolean);
    expect(tickers).toContain("NVDA");
    expect(tickers).not.toContain("MSFT");
  });
});
