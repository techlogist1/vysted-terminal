import asyncio, sys, os
sys.path.insert(0, os.getcwd())
from services.search import searxng
from services.agent_tools import web_search
from services.research import deep
payload = {"results": [
  {"url": "https://www.livemint.com/market/stock-market-news/infosys-q2-results-2026.html", "title": "Infosys Q2 results beat estimates", "content": "Infosys reported net profit ...", "publishedDate": "2026-09-18T14:05:00"},
  {"url": "https://economictimes.indiatimes.com/markets/stocks/news/infosys-buyback.cms", "title": "Infosys buyback record date", "content": "Infosys board ...", "publishedDate": None},
]}
class B:
    async def search(self, q, options=None):
        from services.search.base import SearchResponse, normalize_results_to_citations
        rs = searxng._map_results(payload)
        return SearchResponse(backend="searxng", query=q, results=rs, citations=normalize_results_to_citations(rs))
async def main():
    res = await web_search._dispatch(B(), "Infosys results", 5, "news", "IN")
    print([ (c["domain"], c["published_at"]) for c in res["citations"]])
    f = deep._Findings(); deep._record_web(f, res, query="Infosys results")
    print([ (s.domain, s.published_at) for s in f.web_sources])
    print([ {k: v for k, v in s.to_dict().items() if k in ("domain","published_at")} for s in f.web_sources])
asyncio.run(main())
