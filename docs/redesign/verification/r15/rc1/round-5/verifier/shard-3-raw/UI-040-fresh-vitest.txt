describe("vs3 fresh UI-040", () => {
  beforeEach(() => resetAgentRunsStoreForTests());
  afterEach(() => {
    stopDelegatePolling();
    vi.restoreAllMocks();
  });

  it("A: a PAUSED run (with its question) is adopted, then the poller syncs it to done", async () => {
    vi.stubGlobal(
      "fetch",
      vi.fn().mockResolvedValue(
        jsonResponse({
          runs: [
            { id: "run-p", agent_id: "researcher", agent_name: "Researcher", status: "paused", question: "Which fiscal year?", cost: { tokens: 900, spend_usd: 0, steps: 1 } },
            { id: "run-c", agent_id: "copilot", status: "cancelled" },
          ],
        }),
      ),
    );
    await adoptSidecarRuns();
    let runs = useAgentRunsStore.getState().runs;
    expect(runs).toHaveLength(1);
    expect(runs[0]).toMatchObject({ sidecarRunId: "run-p", status: "paused", question: "Which fiscal year?" });

    vi.stubGlobal(
      "fetch",
      vi.fn().mockResolvedValue(
        jsonResponse({ runs: [{ id: "run-p", agent_id: "researcher", status: "done", cost: { tokens: 3000, spend_usd: 0, steps: 3 } }] }),
      ),
    );
    await pollDelegateRuns();
    runs = useAgentRunsStore.getState().runs;
    expect(runs[0].status).toBe("done");
  });

  it("B: a cancel whose request never reaches the sidecar (network throw) leaves the run running", async () => {
    const id = useAgentRunsStore.getState().startRun({ agentId: "copilot", agentName: "Copilot", mode: "delegate", sidecarRunId: "run-n" });
    vi.stubGlobal("fetch", vi.fn().mockRejectedValue(new TypeError("Failed to fetch")));
    const r = await cancelDelegateRun("run-n");
    expect(r.ok).toBe(false);
    expect(r.error).toMatch(/retry/);
    expect(useAgentRunsStore.getState().runs.find((x) => x.id === id)?.status).toBe("running");
  });
});
