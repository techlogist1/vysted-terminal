describe("vs3 fresh RESEARCH-026", () => {
  it("A: EUR and JPY market caps read exactly as Equity Overview's formatCompactMoney", () => {
    for (const ccy of ["EUR", "JPY"]) {
      const m = deriveMetrics(
        structured("equity", {
          price: { ok: true, provider: "yfinance", data: quote({ currency: ccy }) },
        }),
      );
      const card = m?.items.find((i) => i.label === "Market cap");
      expect(card?.value).toBe(formatCompactMoney(2_000_000_000, ccy));
    }
  });

  it("B: unknown currency is stated on revenue and on a currency-unit derived card too", () => {
    const m = deriveMetrics(
      structured("equity", {
        price: undefined,
        fundamentals: { ok: true, provider: "yfinance", data: fundamentals({ currency: null }) },
        derived: derivedLeg({
          dividend_per_share: { value: 12.5, label: "Dividend / share", unit: "currency" },
        } as unknown as BriefDerivedMetrics),
      }),
    );
    const byLabel = Object.fromEntries((m?.items ?? []).map((i) => [i.label, i.value]));
    expect(byLabel["Revenue"]).toMatch(/currency unknown/);
    expect(byLabel["Dividend / share"]).toMatch(/currency unknown/);
    // never byte-identical to the USD rendering
    const usd = deriveMetrics(structured("equity", {}));
    const usdRev = usd?.items.find((i) => i.label === "Revenue")?.value;
    expect(byLabel["Revenue"]).not.toBe(usdRev);
  });
});
