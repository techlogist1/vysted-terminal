describe("vs3 fresh AGENT-041", () => {
  beforeEach(() => resetProposedChangesStoreForTests());

  it("A: an accepted write_note REPLACE on a symbol note is undone to the original text", async () => {
    useNotesStore.getState().setSymbolNote("INFY", "original thesis: margin recovery");
    const { id } = useProposedChangesStore.getState().enqueue({
      toolCallId: "t1", name: "write_note", input: { scope: "INFY", mode: "replace", text: "agent rewrite" }, batchId: "b1",
    });
    expect(await useProposedChangesStore.getState().accept(id)).toBe("applied");
    expect(useNotesStore.getState().noteFor("INFY")).toBe("agent rewrite");
    useProposedChangesStore.getState().undo(id);
    expect(useNotesStore.getState().noteFor("INFY")).toBe("original thesis: margin recovery");
    expect(useProposedChangesStore.getState().changes.find((c) => c.id === id)?.status).toBe("undone");
  });

  it("B: an accepted portfolio_update_position is undone to the prior lot", async () => {
    const pid = "p1";
    usePortfoliosStore.getState().setAll(
      [{ id: pid, name: "Main", holdings: [
        { id: "h1", symbol: "TCS", quantity: 10, costBasis: 3500, assetClass: "equity" },
        { id: "h2", symbol: "HDFCBANK", quantity: 5, costBasis: 1600, assetClass: "equity" },
      ] }],
      pid,
    );
    const { id } = useProposedChangesStore.getState().enqueue({
      toolCallId: "t2", name: "portfolio_update_position", input: { symbol: "TCS", quantity: 25 }, batchId: "b2",
    });
    expect(await useProposedChangesStore.getState().accept(id)).toBe("applied");
    const after = usePortfoliosStore.getState().portfolios[0].holdings.find((h) => h.symbol === "TCS");
    expect(after?.quantity).toBe(25);
    useProposedChangesStore.getState().undo(id);
    const holdings = usePortfoliosStore.getState().portfolios[0].holdings;
    expect(holdings.map((h) => [h.id, h.symbol, h.quantity, h.costBasis])).toEqual([
      ["h1", "TCS", 10, 3500],
      ["h2", "HDFCBANK", 5, 1600],
    ]);
  });
});
