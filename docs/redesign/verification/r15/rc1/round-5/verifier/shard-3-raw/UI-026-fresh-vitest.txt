describe("vs3 fresh UI-026", () => {
  it("A: a symbol added during an in-flight poll survives that poll landing without it", async () => {
    vi.useFakeTimers();
    mockFetch.mockResolvedValue(rowsFor(DEFAULT_SYMBOLS, "live", "REGULAR"));
    render(<WatchlistPanel />);
    await act(async () => {
      await vi.advanceTimersByTimeAsync(0);
    });
    let resolvePoll: (rows: WatchlistRow[]) => void = () => {};
    mockFetch.mockReturnValueOnce(new Promise((r) => (resolvePoll = r)));
    await act(async () => {
      await vi.advanceTimersByTimeAsync(5_000);
    });
    act(() => useSymbolsStore.getState().addSymbol("AMZN", "equity"));
    await act(async () => resolvePoll(rowsFor(DEFAULT_SYMBOLS, "live", "REGULAR")));
    expect(screen.getByText("AMZN")).toBeInTheDocument();
  });

  it("B: 'INF' candidates still showing when 'INFY' + Enter adds INFY, not INF's top pick", async () => {
    autocompleteMock.mockImplementation(async (_p: string, params: { q: string }) =>
      params.q === "INF"
        ? {
            query: "INF",
            region: "IN",
            candidates: [
              { symbol: "INFIBEAM", name: "Infibeam Avenues", exchange: "NSE", region: "IN", asset_class: "equity", yahoo_symbol: "INFIBEAM.NS", confidence: 0.8 },
            ],
          }
        : new Promise(() => {}),
    );
    render(<WatchlistPanel />);
    const input = screen.getByLabelText("Add symbol");
    fireEvent.change(input, { target: { value: "INF" } });
    expect(await screen.findByRole("option", { name: /Infibeam/ })).toBeInTheDocument();
    fireEvent.keyDown(input, { key: "ArrowDown" });
    fireEvent.change(input, { target: { value: "INFY" } });
    fireEvent.submit(input.closest("form")!);
    const symbols = useSymbolsStore.getState().entries.map((e) => e.symbol);
    expect(symbols).toContain("INFY");
    expect(symbols).not.toContain("INFIBEAM");
  });

  it("C: backoff doubles on repeated failures and resets after a success", async () => {
    vi.useFakeTimers();
    mockFetch.mockReset();
    mockFetch
      .mockRejectedValueOnce(new SidecarError(502, "d1"))
      .mockRejectedValueOnce(new SidecarError(502, "d2"))
      .mockResolvedValue(rowsFor(DEFAULT_SYMBOLS, "live", "REGULAR"));
    render(<WatchlistPanel />);
    await act(async () => { await vi.advanceTimersByTimeAsync(0); });
    expect(mockFetch).toHaveBeenCalledTimes(1);
    await act(async () => { await vi.advanceTimersByTimeAsync(10_000); });
    expect(mockFetch).toHaveBeenCalledTimes(2); // 10 s after 1 failure
    await act(async () => { await vi.advanceTimersByTimeAsync(19_000); });
    expect(mockFetch).toHaveBeenCalledTimes(2); // 20 s after 2 failures
    await act(async () => { await vi.advanceTimersByTimeAsync(1_000); });
    expect(mockFetch).toHaveBeenCalledTimes(3); // success
    await act(async () => { await vi.advanceTimersByTimeAsync(5_000); });
    expect(mockFetch).toHaveBeenCalledTimes(4); // back to 5 s
  });
});
