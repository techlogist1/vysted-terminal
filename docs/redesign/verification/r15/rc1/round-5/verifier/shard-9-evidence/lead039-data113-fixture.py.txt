            "currency": cur, "financial_currency": fin, "total_revenue": rev, "trailing_eps": teps, "country": country, "name": sym}

ep._fetch_calendar_sync = lambda s: CASES[s]
async def main():
    for s in CASES:
        try:
            d = (await ep.get_estimate_detail(s)).model_dump(mode="json")
            print(s, json.dumps({k: d.get(k) for k in ("eps_estimate_mean","eps_estimate_high","eps_estimate_low","revenue_estimate_mean","currency","revenue_currency")}))
        except Exception as e:
            print(s, "ERR", type(e).__name__, e)
asyncio.run(main())
