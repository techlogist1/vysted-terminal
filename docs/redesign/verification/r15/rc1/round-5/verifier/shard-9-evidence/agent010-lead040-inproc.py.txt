import asyncio, os, socket, sys, threading, time

async def ticker(stop, lags):
    while not stop.is_set():
        t = time.perf_counter(); await asyncio.sleep(0.01); lags.append(time.perf_counter() - t - 0.01)

async def with_lag(coro):
    stop = asyncio.Event(); lags = []
    tk = asyncio.create_task(ticker(stop, lags))
    t = time.perf_counter(); res = await coro; dt = time.perf_counter() - t
    stop.set(); await tk
    return res, dt, max(lags) if lags else 0.0

def blackhole():
    s = socket.socket(); s.bind(("127.0.0.1", 0)); s.listen(50); held = []
    def run():
        while True:
            c, _ = s.accept(); held.append(c)
    threading.Thread(target=run, daemon=True).start()
    return s.getsockname()[1]
