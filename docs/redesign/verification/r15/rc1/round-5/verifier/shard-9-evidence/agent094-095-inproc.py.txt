# catalog walk: every schema with a list-valued type anywhere
def walk(s, path, out):
    if isinstance(s, dict):
        if isinstance(s.get('type'), list): out.append(path)
        for k, v in s.items(): walk(v, path+'.'+k, out)
    elif isinstance(s, list):
        for i, v in enumerate(s): walk(v, path+f'[{i}]', out)
