import asyncio, sys
from typing import Any
import pytest
sys.path.insert(0, "tests")
from test_run_manager import _await_terminal, _patch, _isolated  # noqa: F401  (autouse fixture)
from models.llm import LLMDoneEvent, LLMToolUseEvent, LLMUsage
from models.run import RunBudget
from services import agent_runtime, run_manager
from services.agent_tools.schemas import HOST_ACTION_TOOLS

print("HOST_ACTION_TOOLS", sorted(HOST_ACTION_TOOLS))

class _Scripted:
    def __init__(self, rounds): self.rounds = rounds; self.calls = 0
    async def stream_chat(self, messages, model, api_key=None, **kw):
        r = self.rounds[min(self.calls, len(self.rounds) - 1)]; self.calls += 1
        for ev in r["events"]:
            yield ev
        if r.get("hang"):
            await asyncio.sleep(30)
        yield LLMDoneEvent(usage=LLMUsage(input_tokens=r.get("tokens", 5), output_tokens=0))

def _tu(i, name, inp): return LLMToolUseEvent(tool_call_id=i, name=name, input=inp)
NOTE = {"scope": "NVDA", "text": "Watch the margin"}
BRIEF = {"title": "halted brief", "markdown": "## B"}

async def _run(monkeypatch, provider, budget):
    _patch(monkeypatch, provider); dispatched = []
    async def _rec(tool_call, *_a, **_k):
        dispatched.append(tool_call.name); return '{"ok": true}'
    monkeypatch.setattr(agent_runtime, "_dispatch_tool", _rec)
    rid = run_manager.launch_run(agent_id="copilot", prompt="x", api_key="sk", budget=budget)
    row = await _await_terminal(rid, timeout=8.0)
    names = [a.get("name") if isinstance(a, dict) else getattr(a, "name", a) for a in (row.host_actions or [])]
    print(f"\nV9 status={row.status} detail={row.detail!r} dispatched={dispatched} host_actions={names} brief={(row.brief or {}).get('title')}")
    return row, dispatched, names

@pytest.mark.asyncio
async def test_wall_halt_mid_round_after_tool_use(monkeypatch):
    row, d, names = await _run(monkeypatch, _Scripted([{"events": [_tu("c1", "write_note", NOTE), _tu("c2", "publish_brief", BRIEF)], "hang": True}]), RunBudget(max_wall_seconds=0.4))
    assert row.status == "error" and names == [] and row.brief is None

@pytest.mark.asyncio
async def test_prior_round_kept_halted_round_dropped(monkeypatch):
    rounds = [{"events": [_tu("c1", "add_to_watchlist", {"symbol": "NVDA"})], "tokens": 5},
              {"events": [_tu("c2", "write_note", NOTE), _tu("c3", "publish_brief", BRIEF)], "tokens": 100_000}]
    row, d, names = await _run(monkeypatch, _Scripted(rounds), RunBudget(max_tokens=1000))
    assert row.status == "error" and names == [n for n in d if n in HOST_ACTION_TOOLS] and row.brief is None

@pytest.mark.asyncio
async def test_spend_halt(monkeypatch):
    row, d, names = await _run(monkeypatch, _Scripted([{"events": [_tu("c1", "write_note", NOTE)], "tokens": 5_000_000}]), RunBudget(max_spend_usd=0.0001))
    assert row.status == "error" and names == [] 

@pytest.mark.asyncio
async def test_step_halt(monkeypatch):
    row, d, names = await _run(monkeypatch, _Scripted([{"events": [_tu("c1", "write_note", NOTE)], "tokens": 1}]), RunBudget(max_steps=1))
    assert set(names) <= set(d)
