import asyncio, os, tempfile
from datetime import date
from pathlib import Path
import httpx
from services import option_chain, nse_bhavcopy, data_cache

ROWS = {"NIFTY": [["2026-09-29", 23000, "call", 100, 10, 1.0, 1.0, 5, 23000.0]]}

async def scenario(label, today, outcomes, cached_days, calls_n=3):
    data_cache.reset_for_tests(Path(tempfile.mkdtemp()) / "c.db")
    nse_bhavcopy._ist_today = lambda: today
    calls = []
    async def fake(day):
        calls.append(day.isoformat()); return outcomes.get(day.isoformat(), ("missing", None))
    option_chain._fetch_fo_day = fake
    for d in cached_days:
        await data_cache.set(option_chain._cache_key(date.fromisoformat(d)), {"rows": ROWS})
    res = []
    for _ in range(calls_n):
        r = await option_chain.fetch_latest_fo()
        res.append(r.trade_date.isoformat() if r else None)
    print(f"{label}: results={res} network_calls={calls}")

async def truncated_zip():
    data_cache.reset_for_tests(Path(tempfile.mkdtemp()) / "c.db")
    nse_bhavcopy._ist_today = lambda: date(2026, 9, 22)
    await data_cache.set(option_chain._cache_key(date(2026, 9, 21)), {"rows": ROWS})
    orig = option_chain._fetch_fo_day
    async def fake_get(url, *a, **k):
        return httpx.Response(200, content=b"PK\x03\x04truncated-download")
    nse_bhavcopy._throttled_get = fake_get
    try:
        r = await option_chain.fetch_latest_fo()
        print("TRUNCATED-ZIP today, Mon cached ->", r.trade_date if r else None)
    except Exception as e:
        print("TRUNCATED-ZIP today, Mon cached -> RAISED", type(e).__name__, e)

async def main():
    real = option_chain._fetch_fo_day
    await scenario("C acceptance (Tue missing, Mon failed, Fri cached)", date(2026, 9, 22), {"2026-09-22": ("missing", None), "2026-09-21": ("failed", None)}, ["2026-09-18"])
    await scenario("D fresh (Wed failed, Tue cached)", date(2026, 9, 23), {"2026-09-23": ("failed", None)}, ["2026-09-22"])
    await scenario("E fresh (Thu missing, Wed missing, Tue failed, Mon uncached, Fri cached)", date(2026, 9, 24), {"2026-09-24": ("missing", None), "2026-09-23": ("missing", None), "2026-09-22": ("failed", None)}, ["2026-09-18"])
    await scenario("F fresh (Mon failed, nothing cached -> honest None)", date(2026, 9, 21), {"2026-09-21": ("failed", None)}, [])
    option_chain._fetch_fo_day = real
    await truncated_zip()
asyncio.run(main())
