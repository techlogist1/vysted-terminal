import sys, asyncio; sys.path.insert(0,'.')
import ollama
from models.llm import LLMDeltaEvent, LLMToolUseEvent

from services.llm.ollama import OllamaProvider

def mk(chunks_by_round):
    class Fake:
        calls=0
        def __init__(self,**_): pass
        async def chat(self, **kw):
            i=Fake.calls; Fake.calls+=1
            chunks=chunks_by_round[min(i,len(chunks_by_round)-1)]
            async def it():
                for c in chunks: yield {"message":{"content":c},"done":False}
                yield {"message":{"content":""},"done":True,"done_reason":"stop","prompt_eval_count":1,"eval_count":1}
            return it()
        async def list(self): return {"models":[{"model":"llama3.1:8b"}]}
    return Fake

async def canned(name, args):
    return {"ok": True, "symbol": args.get("symbol"), "name": "Sify Technologies", "market_cap": 1.0e8}

async def run(label, rounds):
    agent_runtime.reload()
    ollama.AsyncClient = mk(rounds)
    prov = OllamaProvider()
    agent_runtime.get_provider = lambda *_a, **_k: prov
    agent_runtime._ACK_GRACE_SECONDS = 0.0
    agent_tools.invoke_tool = canned
    ev=[e async for e in agent_runtime.invoke_agent(agent_id="copilot", prompt="What is SIFY's ADR ratio?", api_key=None, provider="ollama", model="llama3.1:8b", mode="ask", autonomy="ask")]
    text="".join(e.text for e in ev if isinstance(e,LLMDeltaEvent))
    calls=[(e.name,e.input) for e in ev if isinstance(e,LLMToolUseEvent)]
    print(f"== {label}\n tool_calls={calls}\n streamed_text={text!r}\n")
